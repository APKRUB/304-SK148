function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var contents = JSON.parse(e.postData.contents); var record = contents.record; var createdAt = record.created_at ? new Date(record.created_at) : new Date(); var imageFormula = '=IMAGE("' + record.image_url + '")'; sheet.appendRow([ record.id, record.name, record.role, record.impression, imageFormula, // 🖼️ แสดงเป็นรูปภาพในเซลล์ record.image_url, // 🔗 ลิงก์รูปภาพ createdAt ]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
